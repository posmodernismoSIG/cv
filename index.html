Geospatial Engineering

Most geospatial problems come down to three things: store spatial data so queries stay fast, process large rasters without loading them into memory, and serve maps that stay responsive in the browser.

My default toolkit is open source: PostGIS for spatial SQL, GDAL, Rasterio, GeoPandas and Shapely for processing, STAC catalogs and Cloud Optimized GeoTIFFs on object storage for imagery, and MapLibre GL JS and deck.gl on the frontend.

Long-running work such as raster processing and provider downloads runs in background workers, so API endpoints stay fast and predictable.

GeoJSON from PostGIS with FastAPIAsync endpoint, connection pool, Pydantic models, bbox validationPython

Returns detections inside a bounding box as a GeoJSON FeatureCollection. Credentials come from the environment, queries are parameterized, and every query has a timeout.

import json
import os
from contextlib import asynccontextmanager
from datetime import date
from typing import Literal

import asyncpg
from fastapi import FastAPI, HTTPException, Query, Request
from pydantic import BaseModel


class DetectionProps(BaseModel):
    area_ha: float
    detected_on: date


class Feature(BaseModel):
    type: Literal["Feature"] = "Feature"
    id: int
    geometry: dict
    properties: DetectionProps


class FeatureCollection(BaseModel):
    type: Literal["FeatureCollection"] = "FeatureCollection"
    features: list[Feature]


@asynccontextmanager
async def lifespan(app: FastAPI):
    # Credentials come from the environment, never from source code
    app.state.pool = await asyncpg.create_pool(
        os.environ["DATABASE_URL"], min_size=1, max_size=10, command_timeout=10
    )
    yield
    await app.state.pool.close()


app = FastAPI(lifespan=lifespan)

# detections.geom: geometry(MultiPolygon, 4326) with a GiST index
SQL = """
    SELECT id, area_ha, detected_on, ST_AsGeoJSON(geom, 6) AS geometry
    FROM detections
    WHERE ST_Intersects(geom, ST_MakeEnvelope($1, $2, $3, $4, 4326))
    ORDER BY detected_on DESC
    LIMIT $5
"""


def parse_bbox(bbox: str) -> tuple[float, float, float, float]:
    try:
        west, south, east, north = (float(v) for v in bbox.split(","))
    except ValueError:
        raise HTTPException(422, "bbox must be 'west,south,east,north'") from None
    if not (-180 <= west < east <= 180 and -90 <= south < north <= 90):
        raise HTTPException(422, "bbox is outside WGS84 bounds")
    return west, south, east, north


@app.get("/detections", response_model=FeatureCollection)
async def list_detections(
    request: Request,
    bbox: str = Query(..., examples=["-73.5,1.0,-71.5,3.0"]),
    limit: int = Query(500, ge=1, le=5000),
) -> FeatureCollection:
    rows = await request.app.state.pool.fetch(SQL, *parse_bbox(bbox), limit)
    return FeatureCollection(
        features=[
            Feature(
                id=row["id"],
                geometry=json.loads(row["geometry"]),
                properties=DetectionProps(
                    area_ha=row["area_ha"], detected_on=row["detected_on"]
                ),
            )
            for row in rows
        ]
    )
Sentinel-2 NDVI to Cloud Optimized GeoTIFFRasterio · Red = B04, NIR = B08 · nodata handling · COG outputPython

Computes NDVI from Sentinel-2 L2A bands, applies the reflectance offset, masks nodata, and writes a compressed COG with internal overviews that TiTiler can serve directly.

import numpy as np
import rasterio

# Sentinel-2 L2A: reflectance = (DN + offset) / 10000.
# Offset is -1000 for processing baseline >= 04.00 (Jan 2022), 0 before.
QUANTIFICATION = 10_000


def ndvi_to_cog(b04_path: str, b08_path: str, out_path: str, offset: int = -1000) -> None:
    """NDVI = (NIR - Red) / (NIR + Red) with B08 (NIR) and B04 (Red), both 10 m."""
    with rasterio.open(b04_path) as red_src, rasterio.open(b08_path) as nir_src:
        if red_src.transform != nir_src.transform or red_src.shape != nir_src.shape:
            raise ValueError("B04 and B08 must share the same 10 m grid")
        red_dn, nir_dn = red_src.read(1), nir_src.read(1)
        crs, transform = red_src.crs, red_src.transform

    def reflectance(dn: np.ndarray) -> np.ndarray:
        return np.clip((dn.astype("float32") + offset) / QUANTIFICATION, 0, None)

    red, nir = reflectance(red_dn), reflectance(nir_dn)
    total = nir + red
    valid = (red_dn > 0) & (nir_dn > 0) & (total > 0)  # DN 0 = no data

    ndvi = np.full(red.shape, np.nan, dtype="float32")
    ndvi[valid] = (nir[valid] - red[valid]) / total[valid]

    profile = {
        "driver": "COG",  # GDAL >= 3.1: tiled, with internal overviews
        "width": ndvi.shape[1],
        "height": ndvi.shape[0],
        "count": 1,
        "dtype": "float32",
        "nodata": np.nan,
        "crs": crs,
        "transform": transform,
        "compress": "DEFLATE",
        "predictor": "YES",
        "blocksize": 512,
        "overview_resampling": "average",
    }
    with rasterio.open(out_path, "w", **profile) as dst:
        dst.write(ndvi, 1)


# Inputs can be local files or remote COG hrefs from a Sentinel-2 L2A STAC item
ndvi_to_cog("B04.tif", "B08.tif", "ndvi_cog.tif")
Vector tiles in React with MapLibre GL JSNext.js client component · vector tile source · cleanup on unmountTypeScript

Loads a vector tile layer (for example from Martin over PostGIS) and removes the map on unmount so the WebGL context isn't leaked.

"use client";

import { useEffect, useRef } from "react";
import maplibregl from "maplibre-gl";
import "maplibre-gl/dist/maplibre-gl.css";

type Props = {
  tilesUrl: string;    // e.g. "https://tiles.example.com/detections/{z}/{x}/{y}"
  sourceLayer: string; // layer name inside the vector tiles, e.g. "detections"
};

export default function DetectionsMap({ tilesUrl, sourceLayer }: Props) {
  const containerRef = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!containerRef.current) return;

    const map = new maplibregl.Map({
      container: containerRef.current,
      style: "https://demotiles.maplibre.org/style.json",
      center: [-72.6, 2.0], // lng, lat
      zoom: 6,
    });
    map.addControl(new maplibregl.NavigationControl(), "top-right");

    map.on("load", () => {
      map.addSource("detections", {
        type: "vector",
        tiles: [tilesUrl],
        minzoom: 4,
        maxzoom: 14,
      });
      map.addLayer({
        id: "detections-fill",
        type: "fill",
        source: "detections",
        "source-layer": sourceLayer,
        paint: { "fill-color": "#ef476f", "fill-opacity": 0.4 },
      });
      map.addLayer({
        id: "detections-outline",
        type: "line",
        source: "detections",
        "source-layer": sourceLayer,
        paint: { "line-color": "#ef476f", "line-width": 1 },
      });
    });

    return () => map.remove(); // release the WebGL context on unmount
  }, [tilesUrl, sourceLayer]);

  return (
    <div
      ref={containerRef}
      role="region"
      aria-label="Deforestation detections map"
      style={{ width: "100%", height: 480 }}
    />
  );
}
